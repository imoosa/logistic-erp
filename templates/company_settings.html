📖 Integration Guide
Endpoint: GET https://eship.magnustic.com/api/v1/track/<docket_no>
Header: X-API-Key: <your key>
⚠️ This call must happen on your server — never in browser JavaScript.
define( 'MAGNUSTIC_API_KEY', 'YOUR_KEY_HERE' );
Install magnustic-tracking.php plugin, then add this to wp-config.php. Use [track_shipment] shortcode on any page.
$ch = curl_init("https://eship.magnustic.com/api/v1/track/" . urlencode($docket_no));
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, ["X-API-Key: YOUR_KEY_HERE"]);
$response = json_decode(curl_exec($ch), true);
curl_close($ch);
import requests

resp = requests.get(
    f"https://eship.magnustic.com/api/v1/track/{docket_no}",
    headers={"X-API-Key": "YOUR_KEY_HERE"},
    timeout=10,
)
data = resp.json()
// Node.js backend only — never run this in browser JS
const resp = await fetch(
  `https://eship.magnustic.com/api/v1/track/${docketNo}`,
  { headers: { "X-API-Key": "YOUR_KEY_HERE" } }
);
const data = await resp.json();
